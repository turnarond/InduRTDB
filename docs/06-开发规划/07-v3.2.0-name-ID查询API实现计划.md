# InduRTDB v3.2.0 name→ID 查询 API 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 InduRTDB 新增 name→ID / name→元数据 查询 API（进程内查找表，布局零改动），作为 node-server 集成的前置依赖。

**Architecture:** `indurtdb_load_config()` 解析 YAML 点位元数据时，把 `irt_point_meta_batch_t.points` 的所有权转移到单例 `g_rtdb` 的进程内只读表 `meta_table`；新增两个公共 API 对该表做线性查找（仅注册期调用，非热路径，线性即可）。共享内存布局不变。

**Tech Stack:** 纯 C11（库）、C++17+gtest（测试）、`-Wall -Wextra -Werror`。

**Spec:** `docs/03-设计文档/04-node-server集成设计文档.md`（第 4 节"InduRTDB v3.2.0 新增 API"）

## Global Constraints

- 库为纯 C11，`-Wall -Wextra -Werror` 零警告；测试 C++17+gtest。
- 共享内存布局（128B/64B/16B、magic、shm version）**不得改动**；本计划只加进程内结构。

```cpp
/**
 * @file test_c_name_lookup.cpp
 * @brief name→ID / name→元数据 查询 API 单元测试
 * @version 3.2.0
 */
#include <gtest/gtest.h>
#include <indurtdb/indurtdb.h>
#include <cstdio>
#include <cstring>

namespace {

class NameLookupTest : public ::testing::Test {
protected:
    void SetUp() override { indurtdb_shutdown(); }
    void TearDown() override { indurtdb_shutdown(); }

    /* 写一份含 3 个点位的 YAML 点位配置 */
    void WritePointConfig(const char* path) {
        FILE* f = std::fopen(path, "w");
        ASSERT_NE(f, nullptr);
        std::fprintf(f, "points:\n");
        std::fprintf(f, "  - id: 1001\n    name: \"AHU_01.Supply_Temp\"\n"
                        "    type: double\n    unit: 1\n    access: 3\n");
        std::fprintf(f, "  - id: 2002\n    name: \"AHU_01.Fan_State\"\n"
                        "    type: bool\n    unit: 0\n    access: 1\n");
        std::fprintf(f, "  - id: 3003\n    name: \"AHU_01.Count\"\n"
                        "    type: int32\n    unit: 2\n    access: 3\n");
        std::fclose(f);
    }
};

TEST_F(NameLookupTest, GetIdByNameHit) {
    ASSERT_EQ(indurtdb_initialize("nl_hit", 4096, 8), 0);
    WritePointConfig("/tmp/nl_hit.yaml");
    ASSERT_EQ(indurtdb_load_config("/tmp/nl_hit.yaml"), 0);
    EXPECT_EQ(indurtdb_get_id_by_name("AHU_01.Supply_Temp"), 1001);
    EXPECT_EQ(indurtdb_get_id_by_name("AHU_01.Fan_State"), 2002);
    EXPECT_EQ(indurtdb_get_id_by_name("AHU_01.Count"), 3003);
    std::remove("/tmp/nl_hit.yaml");
}

TEST_F(NameLookupTest, GetIdByNameMiss) {
    ASSERT_EQ(indurtdb_initialize("nl_miss", 4096, 8), 0);
    WritePointConfig("/tmp/nl_miss.yaml");
    ASSERT_EQ(indurtdb_load_config("/tmp/nl_miss.yaml"), 0);
    EXPECT_EQ(indurtdb_get_id_by_name("Nonexistent.Point"), -1);
    std::remove("/tmp/nl_miss.yaml");
}

TEST_F(NameLookupTest, GetMetaByName) {
    ASSERT_EQ(indurtdb_initialize("nl_meta", 4096, 8), 0);
    WritePointConfig("/tmp/nl_meta.yaml");
    ASSERT_EQ(indurtdb_load_config("/tmp/nl_meta.yaml"), 0);
    uint8_t type = 0xff; uint16_t unit = 0; uint8_t access = 0;
    EXPECT_EQ(indurtdb_get_meta_by_name("AHU_01.Supply_Temp", &type, &unit, &access), 0);
    EXPECT_EQ(type, INDURTDB_TYPE_DOUBLE);
    EXPECT_EQ(unit, 1);
    EXPECT_EQ(access, 3);
    std::remove("/tmp/nl_meta.yaml");
}

TEST_F(NameLookupTest, BeforeLoadConfigReturnsError) {
    ASSERT_EQ(indurtdb_initialize("nl_nocfg", 4096, 8), 0);
    EXPECT_EQ(indurtdb_get_id_by_name("AHU_01.Supply_Temp"), -1);
}

TEST_F(NameLookupTest, UninitializedReturnsError) {
    EXPECT_EQ(indurtdb_get_id_by_name("any"), -1);
    uint8_t t; uint16_t u; uint8_t a;
    EXPECT_EQ(indurtdb_get_meta_by_name("any", &t, &u, &a), -1);
}

TEST_F(NameLookupTest, NullOrEmptyNameReturnsError) {
    ASSERT_EQ(indurtdb_initialize("nl_null", 4096, 8), 0);
    WritePointConfig("/tmp/nl_null.yaml");
    ASSERT_EQ(indurtdb_load_config("/tmp/nl_null.yaml"), 0);
    EXPECT_EQ(indurtdb_get_id_by_name(nullptr), -1);
    EXPECT_EQ(indurtdb_get_id_by_name(""), -1);
    std::remove("/tmp/nl_null.yaml");
}

}  // namespace
```

```c
/* ---- name→ID / name→元数据 查询 ---- */

int indurtdb_get_id_by_name(const char* name) {
    if (!__atomic_load_n(&g_rtdb.initialized, __ATOMIC_ACQUIRE)) {
        set_error("not initialized"); return -1;
    }
    if (!name || name[0] == '\0') { set_error("null or empty name"); return -1; }
    if (!g_rtdb.meta_table || g_rtdb.meta_count == 0) {
        set_error("no point metadata loaded"); return -1;
    }
    for (size_t i = 0; i < g_rtdb.meta_count; ++i) {
        if (strncmp(g_rtdb.meta_table[i].name, name,
                    sizeof(g_rtdb.meta_table[i].name)) == 0)
            return (int)g_rtdb.meta_table[i].id;
    }
    set_error("name not found");
    return -1;
}

int indurtdb_get_meta_by_name(const char* name,
                              uint8_t* type, uint16_t* unit, uint8_t* access) {
    if (!__atomic_load_n(&g_rtdb.initialized, __ATOMIC_ACQUIRE)) {
        set_error("not initialized"); return -1;
    }
    if (!name || name[0] == '\0') { set_error("null or empty name"); return -1; }
    if (!g_rtdb.meta_table || g_rtdb.meta_count == 0) {
        set_error("no point metadata loaded"); return -1;
    }
    for (size_t i = 0; i < g_rtdb.meta_count; ++i) {
        if (strncmp(g_rtdb.meta_table[i].name, name,
                    sizeof(g_rtdb.meta_table[i].name)) == 0) {
            if (type)   *type   = g_rtdb.meta_table[i].type;
            if (unit)   *unit   = g_rtdb.meta_table[i].unit;
            if (access) *access = g_rtdb.meta_table[i].access;
            return 0;
        }
    }
    set_error("name not found");
    return -1;
}
```
